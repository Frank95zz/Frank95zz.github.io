\section{Research Experience}
\cventry{EKF with a 9-Axis IMU and a Hydrostatic Pressure Sensor}{Aug 2025}{Independent project, Shanghai Jiao Tong University\quad |\quad Advisor: Prof. Jianping He}
\begin{itemize}
  \item Developed an extended Kalman filter for underwater robot attitude estimation by fusing 9-axis IMU and pressure measurements on the IWIN-FINS platform; implemented the embedded code.
  \item Experiments showed improved stability compared with an IMU-only Mahony algorithm.
\end{itemize}
\cventry{Deadlock Control and Scheduling with Limited Buffers}{Feb 2024 - Jan 2025}{Independent project, Xi'an Jiaotong University\quad |\quad Advisor: Assoc. Prof. Yanxiang Feng}
\begin{itemize}
  \item Studied deadlocks in a two-stage assembly no-wait permutation flowshop with limited buffers; enhanced discrete particle swarm optimization (DPSO) using Q-learning for dynamic evolution strategy selection.
  \item Proposed a two-stage local search (VND+IG) to escape local optima; the algorithm outperformed HSSA, HGA, and ABC in experiments.
\end{itemize}
\cventry{Design and Development of an AI-Powered Oilfield Terminal}{Apr 2024 - Apr 2025}{National Undergraduate Innovation Training Program\quad |\quad Advisor: Cuicui Zhang, XJTU}
\begin{itemize}
  \item The project used convolutional neural networks and optimization to classify oil-well indicator diagrams, estimate operating conditions, fluid level, and production, and support automatic intermittent pumping on an RK3568 embedded device.
  \item Built the dataset of abnormal oil-well indicator diagrams.
\end{itemize}
